import androidx.compose.ui.Modifier

@Composable
internal fun EntrySaveError(
    message: String?,
    modifier: Modifier = Modifier,
) {
    message?.let { Text(it, modifier = modifier, color = MaterialTheme.colorScheme.error) }
}
