package com.chimali.core.bluetooth.util

/**
 * Constants for Bluetooth HID configuration.
 */
object BluetoothHidConstants {
    /**
     * HID Descriptor for a basic FIDO2 / Keyboard device.
     * This is a simplified version. A real FIDO2 descriptor is more specific.
     */
    val FIDO_HID_REPORT_DESCRIPTOR =
        byteArrayOf(
            0x05.toByte(),
            0x01.toByte(), // Usage Page (Generic Desktop)
            0x09.toByte(),
            0x06.toByte(), // Usage (Keyboard)
            0xA1.toByte(),
            0x01.toByte(), // Collection (Application)
            0x05.toByte(),
            0x07.toByte(), //   Usage Page (Keyboard/Keypad)
            0x19.toByte(),
            0xE0.toByte(), //   Usage Minimum (Keyboard Left Control)
            0x29.toByte(),
            0xE7.toByte(), //   Usage Maximum (Keyboard Right GUI)
            0x15.toByte(),
            0x00.toByte(), //   Logical Minimum (0)
            0x25.toByte(),
            0x01.toByte(), //   Logical Maximum (1)
            0x75.toByte(),
            0x01.toByte(), //   Report Size (1)
            0x95.toByte(),
            0x08.toByte(), //   Report Count (8)
            0x81.toByte(),
            0x02.toByte(), //   Input (Data, Variable, Absolute)
            // ... (rest of keyboard descriptor would follow)
            0xC0.toByte(), // End Collection
        )

    // In Chimali, we'd use a specific descriptor for FIDO HID over GATT or Bluetooth Classic HID.
}
